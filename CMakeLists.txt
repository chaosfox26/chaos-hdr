cmake_minimum_required(VERSION 3.20)
project(truehdr_linux VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(TRUEHDR_DEVELOPMENT "Enable development-only controls and fixtures" OFF)
option(TRUEHDR_TESTS_ONLY "Build portable native checks without desktop integration" OFF)
find_package(Qt6 REQUIRED COMPONENTS Core)
if(NOT TRUEHDR_TESTS_ONLY)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6GuiAddons REQUIRED)
add_executable(truehdr-panel native/panel.cpp)
target_link_libraries(truehdr-panel PRIVATE Qt6::Widgets Qt6::DBus KF6::GlobalAccel KF6::GuiAddons)
if(TRUEHDR_DEVELOPMENT)
    target_compile_definitions(truehdr-panel PRIVATE TRUEHDR_DEVELOPMENT)
endif()
endif()
add_executable(truehdr-profiles-test EXCLUDE_FROM_ALL native/test_profiles.cpp)
target_compile_options(truehdr-profiles-test PRIVATE -UNDEBUG)
target_link_libraries(truehdr-profiles-test PRIVATE Qt6::Core)
add_executable(truehdr-overlay-motion-test EXCLUDE_FROM_ALL native/test_overlay_motion.cpp)
target_compile_options(truehdr-overlay-motion-test PRIVATE -UNDEBUG)
