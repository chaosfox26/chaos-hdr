cmake_minimum_required(VERSION 3.25)
project(truehdr_desktop VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_AUTOMOC ON)
option(TRUEHDR_DEVELOPMENT "Enable private fixtures and frame readback diagnostics" OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(Qt6 REQUIRED COMPONENTS Quick)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(Vulkan REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
if(NOT TRUEHDR_ROOT)
    get_filename_component(TRUEHDR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
if(NOT TRUEHDR_ASSET_ROOT)
    set(TRUEHDR_ASSET_ROOT "${TRUEHDR_ROOT}")
endif()
file(GLOB sites "${TRUEHDR_ROOT}/.venv/lib/python*/site-packages")
list(GET sites 0 TRUEHDR_RUNTIME_SITE)
set(TRUEHDR_NGX "${TRUEHDR_ROOT}/output/compositor/reference/ngx")
find_program(GLSLANG glslangValidator REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/desktop_input.spv"
    COMMAND "${GLSLANG}" -V "${CMAKE_CURRENT_SOURCE_DIR}/input.comp" -o "${CMAKE_CURRENT_BINARY_DIR}/desktop_input.spv"
    DEPENDS input.comp)
add_custom_target(desktop_shader DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/desktop_input.spv")
add_library(desktop_gpu STATIC gpu.cpp ../vulkan_hdr.cpp)
set_target_properties(desktop_gpu PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(desktop_gpu PRIVATE "${TRUEHDR_NGX}")
target_link_libraries(desktop_gpu PUBLIC Qt6::Gui Vulkan::Vulkan PkgConfig::EPOXY "${TRUEHDR_NGX}/libnvsdk_ngx.a" ${CMAKE_DL_LIBS})
target_compile_definitions(desktop_gpu PRIVATE TRUEHDR_DEFAULT_PROFILE="${TRUEHDR_ROOT}/profile.json"
    TRUEHDR_RUNTIME_SITE="${TRUEHDR_RUNTIME_SITE}" TRUEHDR_DESKTOP_SHADER="${CMAKE_CURRENT_BINARY_DIR}/desktop_input.spv")
target_compile_options(desktop_gpu PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_dependencies(desktop_gpu desktop_shader)
add_executable(truehdr-desktop-gpu-test gpu_test.cpp)
target_link_libraries(truehdr-desktop-gpu-test PRIVATE desktop_gpu)
find_library(KWIN_LIBRARY kwin REQUIRED)
set(TRUEHDR_EFFECT_ID "truehdr_desktop" CACHE STRING "Native effect plugin ID")
configure_file(effect.json.in "${CMAKE_CURRENT_BINARY_DIR}/effect.json" @ONLY)
add_library(truehdr_desktop MODULE effect.cpp
    ../../vendor/imgui/imgui.cpp ../../vendor/imgui/imgui_draw.cpp
    ../../vendor/imgui/imgui_widgets.cpp ../../vendor/imgui/imgui_tables.cpp
    ../../vendor/imgui/backends/imgui_impl_opengl3.cpp)
target_include_directories(truehdr_desktop PRIVATE /usr/include/kwin ../../vendor/imgui "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(truehdr_desktop PRIVATE desktop_gpu ${KWIN_LIBRARY} Qt6::Widgets Qt6::DBus Qt6::Quick
    KF6::CoreAddons KF6::ConfigCore KF6::WindowSystem PkgConfig::WAYLAND_SERVER PkgConfig::DRM PkgConfig::FONTCONFIG)
target_compile_definitions(truehdr_desktop PRIVATE IMGUI_USE_WCHAR32 TRUEHDR_ROOT="${TRUEHDR_ROOT}" TRUEHDR_ASSET_ROOT="${TRUEHDR_ASSET_ROOT}")
set_target_properties(truehdr_desktop PROPERTIES PREFIX "" OUTPUT_NAME "${TRUEHDR_EFFECT_ID}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/kwin/effects/plugins")
add_executable(truehdr-desktop-window-test window_test.cpp)
target_link_libraries(truehdr-desktop-window-test PRIVATE Qt6::Widgets PkgConfig::EPOXY)
add_executable(truehdr-desktop-session session.cpp)
target_link_libraries(truehdr-desktop-session PRIVATE Qt6::Core Qt6::DBus)
add_library(truehdr_bootstrap MODULE bootstrap.cpp)
target_link_libraries(truehdr_bootstrap PRIVATE Qt6::Quick)
target_compile_definitions(truehdr_bootstrap PRIVATE TRUEHDR_PLUGIN_ROOT="${TRUEHDR_ROOT}/build-desktop")
set_target_properties(truehdr_bootstrap PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bootstrap")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/bootstrap/qmldir" CONTENT "module TrueHDRBootstrap\nplugin truehdr_bootstrap\n")

if(TRUEHDR_DEVELOPMENT)
    find_package(LayerShellQt REQUIRED)
    target_link_libraries(truehdr-desktop-window-test PRIVATE LayerShellQt::Interface)
    target_compile_definitions(truehdr-desktop-window-test PRIVATE TRUEHDR_DEVELOPMENT)
    target_compile_definitions(desktop_gpu PRIVATE TRUEHDR_DEVELOPMENT)
    target_compile_definitions(truehdr_desktop PRIVATE TRUEHDR_DEVELOPMENT)
endif()
